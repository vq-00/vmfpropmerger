cmake_minimum_required(VERSION 3.10)

project(vmfpropmerger)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(SOURCES
    src/main.cpp
    src/utils.cpp
    src/vmf.cpp
    src/smd.cpp
    src/vpk.cpp
    src/options.cpp
    src/transform.cpp
)

add_executable(vmfpropmerger ${SOURCES})

target_include_directories(vmfpropmerger PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/include
)

if(WIN32)
    target_compile_definitions(vmfpropmerger PRIVATE
        _CRT_SECURE_NO_WARNINGS
        NOMINMAX
    )
else()
    target_link_libraries(vmfpropmerger stdc++fs)
endif()

if(MSVC)
    target_compile_options(vmfpropmerger PRIVATE /O2)
else()
    target_compile_options(vmfpropmerger PRIVATE -O2)
endif()

install(TARGETS vmfpropmerger DESTINATION bin)
